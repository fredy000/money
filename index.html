// --- 1. מנוע החישוב החכם (calc) --- function calc() { const today = todayStr(); const thisMonth = monthKey(today); const paidIncomes = (state.incomes || []).filter((i) => i.status === "paid"); const pendingIncomes = (state.incomes || []).filter((i) => i.status !== "paid"); // סך הכנסות שנכנסו בפועל const totalPaidIncomes = paidIncomes.reduce((s, i) => s + Number(i.amount), 0); // הוצאות החודש הנוכחי בלבד const monthExpenses = (state.expenses || []) .filter((e) => monthKey(e.date) === thisMonth) .reduce((s, e) => s + Number(e.amount), 0); // יתרה נזילה עכשווית בבנק const actualCurrentBalance = Number(state.openingBalance || 0) + totalPaidIncomes - monthExpenses; // הכנסות צפויות החודש (שטרם שולמו) const monthExpectedIncome = pendingIncomes .filter((i) => monthKey(i.expectedDate) === thisMonth) .reduce((s, i) => s + Number(i.amount), 0); // תשלומים באיחור (סכום שלא נכנס בזמן) const overdueIncomes = pendingIncomes.filter((i) => i.expectedDate < today); const overdueAmount = overdueIncomes.reduce((s, i) => s + Number(i.amount), 0); // --- לוגיקת המחשבון החכם להוצאה בטוחה (Safe-to-Spend) --- // רזרבת ביטחון לבלת"מים (10% מהיתרה הנזילה או מינימום 500 ש"ח) const safetyBuffer = Math.max(500, actualCurrentBalance * 0.10); // סכום מומלץ להוצאה: יתרה נזילה פחות רזרבת ביטחון let safeToSpend = actualCurrentBalance - safetyBuffer; if (safeToSpend < 0) safeToSpend = 0; // חישוב תחזית לסוף החודש (כולל הכנסות עתידיות צפויות) const finalExpectedBalance = actualCurrentBalance + monthExpectedIncome; return { openingBalance: Number(state.openingBalance || 0), actualCurrentBalance, finalExpectedBalance, monthExpenses, monthExpectedIncome, safeToSpend, safetyBuffer, overdueAmount, overdueCount: overdueIncomes.length }; } // --- 2. תצוגת לוח הבקרה (renderDashboard) --- function renderDashboard() { const c = calc(); return `
💡 מחשבון נזילות חכם
סכום מומלץ להוצאה כעת:
${fmtMoney(c.safeToSpend)}
* מבוסס על יתרה נזילה של ${fmtMoney(c.actualCurrentBalance)} בניכוי רזרבת ביטחון לשעת חירום בסך ${fmtMoney(c.safetyBuffer)}. ${c.overdueCount > 0 ? `
⚠️ שימו לב: קיימים ${c.overdueCount} תשלומים באיחור בסך ${fmtMoney(c.overdueAmount)} (לא נלקחו בחשבון הנזיל).` : ''}
יתרה נזילה בבנק
${fmtMoney(c.actualCurrentBalance)}
הוצאות החודש הנוכחי
${fmtMoney(c.monthExpenses)}
תחזית יתרה לסוף חודש
${fmtMoney(c.finalExpectedBalance)}

ניתוח תזרים מזומנים חודשי

יתרת פתיחה בבנק: ${fmtMoney(c.openingBalance)}
פחות הוצאות החודש הנוכחי: -${fmtMoney(c.monthExpenses)}
פלוס הכנסות שנכנסו בפועל: +${fmtMoney(c.actualCurrentBalance - c.openingBalance + c.monthExpenses)}
יתרה נזילה בפועל (נכון להיום): ${fmtMoney(c.actualCurrentBalance)}
`; }

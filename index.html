function saveCurrentExpenseState() { const descEl = document.getElementById("exp-desc"); const amountEl = document.getElementById("exp-amount"); const dateEl = document.getElementById("exp-date"); const recurringEl = document.getElementById("exp-recurring"); const installmentsCountEl = document.getElementById("exp-installmentsCount"); ui.expenseFormState = { desc: descEl ? descEl.value : ui.expenseFormState.desc, amount: amountEl ? amountEl.value : ui.expenseFormState.amount, date: dateEl ? dateEl.value : ui.expenseFormState.date, category: ui.expenseFormState.category || CATEGORIES[0].name, installmentsCount: installmentsCountEl ? installmentsCountEl.value : ui.expenseFormState.installmentsCount, recurring: recurringEl ? recurringEl.checked : ui.expenseFormState.recurring }; } document.addEventListener("change", (e) => { if (e.target.id === "installments-check") { saveCurrentExpenseState(); ui.isInstallments = e.target.checked; render(); } }); document.addEventListener("click", (e) => { const actionBtn = e.target.closest("[data-action]"); if (!actionBtn) return; const action = actionBtn.dataset.action; if (action === "tab") { ui.tab = actionBtn.dataset.tab; ui.formError = ""; render(); } if (action === "toggle-balance-edit") { ui.showBalanceEdit = !ui.showBalanceEdit; render(); } if (action === "save-balance") { const input = document.getElementById("balance-input"); const val = parseAmount(input ? input.value : 0); if (!isNaN(val)) { // מציאת סך ההוצאות עד היום כדי לעדכן את היתרה הפותחת בצורה מדויקת const today = todayStr(); const expensesTillNow = (state.expenses || []).filter(exp => exp.date <= today).reduce((s, exp) => s + Number(exp.amount), 0); const paidIncomes = (state.incomes || []).filter(inc => inc.status === "paid").reduce((s, inc) => s + Number(inc.amount), 0); // עדכון יתרת הבסיס (Opening Balance) state.openingBalance = val + expensesTillNow - paidIncomes; ui.showBalanceEdit = false; saveToCloud(); } } if (action === "toggle-income-form") { ui.showIncomeForm = !ui.showIncomeForm; ui.editingIncomeId = null; ui.formError = ""; render(); } if (action === "toggle-expense-form") { ui.showExpenseForm = !ui.showExpenseForm; ui.editingExpenseId = null; ui.formError = ""; ui.expenseFormState = { desc: "", amount: "", date: todayStr(), category: CATEGORIES[0].name, installmentsCount: 3, recurring: false }; render(); } if (action === "cancel-edit") { ui.editingIncomeId = null; ui.editingExpenseId = null; ui.showIncomeForm = false; ui.showExpenseForm = false; ui.formError = ""; render(); } if (action === "select-category") { saveCurrentExpenseState(); ui.expenseFormState.category = actionBtn.dataset.cat; render(); } if (action === "mark-paid") { const id = actionBtn.dataset.id; const inc = state.incomes.find(i => i.id === id); if (inc) { inc.status = "paid"; inc.paidDate = todayStr(); saveToCloud(); } } if (action === "edit-income") { ui.tab = "income"; ui.editingIncomeId = actionBtn.dataset.id; ui.showIncomeForm = false; ui.formError = ""; render(); } if (action === "delete-income") { const id = actionBtn.dataset.id; if (confirm("האם למחוק הכנסה זו?")) { state.incomes = state.incomes.filter(i => i.id !== id); saveToCloud(); } } if (action === "edit-expense") { ui.tab = "expenses"; const exp = state.expenses.find(e => e.id === actionBtn.dataset.id); if (exp) { ui.editingExpenseId = exp.id; ui.showExpenseForm = false; ui.expenseFormState = { desc: exp.desc, amount: exp.amount, date: exp.date, category: exp.category || CATEGORIES[0].name, recurring: !!exp.recurring }; ui.formError = ""; render(); } } if (action === "delete-expense") { const id = actionBtn.dataset.id; if (confirm("האם למחוק הוצאה זו?")) { state.expenses = state.expenses.filter(e => e.id !== id); saveToCloud(); } } }); document.addEventListener("submit", (e) => { e.preventDefault(); const form = e.target; if (form.id === "income-form") { const formData = new FormData(form); const id = formData.get("id"); const client = formData.get("client").trim(); const amount = parseAmount(formData.get("amount")); const date = formData.get("date"); const termDays = Number(formData.get("termDays") || 0); if (!client || isNaN(amount) || amount <= 0 || !date) { ui.formError = "נא למלא את כל השדות בצורה תקינה (סכום חיובי)."; render(); return; } if (id) { const inc = state.incomes.find(i => i.id === id); if (inc) { inc.client = client; inc.amount = amount; inc.date = date; inc.termDays = termDays; inc.expectedDate = calcExpectedDate(date, termDays); } } else { state.incomes.push({ id: uid(), client, amount, date, termDays, expectedDate: calcExpectedDate(date, termDays), status: "pending" }); } ui.showIncomeForm = false; ui.editingIncomeId = null; ui.formError = ""; saveToCloud(); } if (form.id === "expense-form") { const formData = new FormData(form); const id = formData.get("id"); const desc = formData.get("desc").trim(); const amount = parseAmount(formData.get("amount")); const date = formData.get("date"); const category = ui.expenseFormState.category || CATEGORIES[0].name; if (!desc || isNaN(amount) || amount <= 0 || !date) { ui.formError = "נא למלא תיאור, סכום תקין ותאריך."; render(); return; } if (id) { const exp = state.expenses.find(e => e.id === id); if (exp) { exp.desc = desc; exp.amount = amount; exp.date = date; exp.category = category; } } else { if (ui.isInstallments) { const count = Math.max(2, Math.min(36, Number(formData.get("installmentsCount") || 3))); const installmentAmount = Math.round((amount / count) * 100) / 100; for (let i = 0; i < count; i++) { state.expenses.push({ id: uid(), desc: `${desc} (${i + 1}/${count})`, amount: installmentAmount, date: addMonths(date, i), category, installmentInfo: `${i + 1}/${count}` }); } } else { const recurring = formData.get("recurring") === "on"; state.expenses.push({ id: uid(), desc, amount, date, category, recurring }); } } ui.showExpenseForm = false; ui.editingExpenseId = null; ui.formError = ""; ui.isInstallments = false; saveToCloud(); } });
